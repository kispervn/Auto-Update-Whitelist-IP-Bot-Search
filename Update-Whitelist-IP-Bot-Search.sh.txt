#!/bin/bash
# How to update new Whitelist IP Bot Search & IP Country database:
# wget https://raw.githubusercontent.com/kispervn/Auto-Update-Whitelist-IP-Bot-Search/main/Update-Whitelist-IP-Bot-Search.sh.txt -O /var/lib/kisper/Update-Whitelist-IP-Bot-Search.sh && echo "12 15 * * 0 root bash /var/lib/kisper/Update-Whitelist-IP-Bot-Search.sh" >> /etc/crontab
#
# Google Bot IP: script tu dong tai 5 file JSON tu Google va cap nhat phan # Google Bot IP Addresses trong whitelist.
# Nguon: https://developers.google.com/crawling/ipranges/
# 1. common-crawlers.json - Googlebot, etc.
# 2. special-crawlers.json - AdsBot, etc.
# 3. user-triggered-fetchers.json - user-triggered fetchers
# 4. user-triggered-fetchers-google.json - Google-triggered fetchers
# 5. user-triggered-agents.json - Google agents

sleep $(( $RANDOM % 200 ));
# Get server IP (try multiple sources for reliability)
ipd=""
for url in "https://api.ipify.org" "http://cpanel.net/showip.cgi"; do
    ipd=$(curl -4sS -L --max-time 10 "$url" 2>/dev/null)
    if echo "$ipd" | grep -qE '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; then
        break
    fi
done
curl -L https://files.voduy.com/vDDoS-Proxy-Protection/conf.d/cdn-ip.conf.txt -o /vddos/conf.d/cdn-ip.conf
# Download whitelist from voduy (overwrite current file)
# Check if downloaded file is valid (not HTML redirect/error)
WHITELIST_DOWNLOADED=0
whitelist_url="https://files.voduy.com/vDDoS-Proxy-Protection/conf.d/whitelist-botsearch.conf.txt"
for attempt in 1 2 3; do
    curl -LsS -o /vddos/conf.d/whitelist-botsearch.conf --max-time 30 "$whitelist_url" 2>/dev/null
    # Kiem tra file hop le: khong phai HTML, bat dau bang "#" hoac so (IP)
    if [ -f /vddos/conf.d/whitelist-botsearch.conf ]; then
        first_char=$(head -c 1 /vddos/conf.d/whitelist-botsearch.conf 2>/dev/null)
        if [ "$first_char" = "#" ] || [[ "$first_char" =~ [0-9] ]]; then
            WHITELIST_DOWNLOADED=1
            break
        fi
    fi
    echo "[WARN] Download attempt $attempt failed, retrying..."
    sleep 2
done
if [ "$WHITELIST_DOWNLOADED" -eq 0 ]; then
    echo "[ERROR] Failed to download valid whitelist file after 3 attempts"
fi

# Merge Google Bot IPs from 5 JSON files into whitelist
# Nguon: https://developers.google.com/crawling/ipranges/
SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
WHITELIST_CONF="/vddos/conf.d/whitelist-botsearch.conf"
GOOGLE_BASE="https://developers.google.com/crawling/ipranges"
GOOGLE_JSON="common-crawlers.json special-crawlers.json user-triggered-fetchers.json user-triggered-fetchers-google.json user-triggered-agents.json"
TMPD="${TMPDIR:-/tmp}/whitelist_google_$$"
mkdir -p "$TMPD"
cleanup() { rm -rf "$TMPD"; }
trap cleanup EXIT

# Tai 4 file JSON tu Google (force refresh - xoa cu, tai moi)
for f in $GOOGLE_JSON; do
    rm -f "$SCRIPT_DIR/$f"
    curl -LsS -o "$SCRIPT_DIR/$f" --max-time 30 "$GOOGLE_BASE/$f" 2>/dev/null || true
    # Kiem tra file co phai JSON hop le (khong phai HTML/redirect)
    if [ -f "$SCRIPT_DIR/$f" ] && head -c 10 "$SCRIPT_DIR/$f" | grep -q '{'; then
        echo "[OK] $f"
    else
        echo "[FAIL] $f - re-downloading..."
        rm -f "$SCRIPT_DIR/$f"
        curl -LsS -o "$SCRIPT_DIR/$f" --max-time 30 -H "Cache-Control: no-cache" "$GOOGLE_BASE/$f?$(date +%s)" 2>/dev/null || true
    fi
done

# Lay tat ca prefix (ipv4/ipv6) tu 5 file JSON
# Cac file JSON:
# https://developers.google.com/crawling/ipranges/common-crawlers.json
# https://developers.google.com/crawling/ipranges/special-crawlers.json
# https://developers.google.com/crawling/ipranges/user-triggered-fetchers.json
# https://developers.google.com/crawling/ipranges/user-triggered-fetchers-google.json
# https://developers.google.com/crawling/ipranges/user-triggered-agents.json
for f in $GOOGLE_JSON; do
    [ -f "$SCRIPT_DIR/$f" ] && grep -oE '"(ipv4|ipv6)Prefix": "[^"]+"' "$SCRIPT_DIR/$f" | sed 's/.*": "//;s/"$//'
done | sort -u > "$TMPD/json_prefixes"

# Lay prefix da co trong phan # Google Bot IP Addresses (den truoc # Bing Bot)
awk '/# Google Bot IP Addresses/{f=1;next} /# Bing Bot IP Addresses/{f=0} f && $2=="yes;" {print $1}' "$WHITELIST_CONF" 2>/dev/null | sort -u > "$TMPD/existing"

# Chi lay prefix chua co (da co thi bo qua)
comm -23 "$TMPD/json_prefixes" "$TMPD/existing" > "$TMPD/new_prefixes"

if [ -s "$TMPD/new_prefixes" ]; then
    NEW_COUNT=$(wc -l < "$TMPD/new_prefixes")
    echo "[INFO] Found $NEW_COUNT new Google Bot prefixes to add"
    # Tim dong bat dau "# Bing Bot IP Addresses"
    BING_LINE=$(grep -n "# Bing Bot IP Addresses" "$WHITELIST_CONF" | head -1 | cut -d: -f1)
    if [ -n "$BING_LINE" ]; then
        # Tao dong "prefix yes;" cho moi prefix moi
        while IFS= read -r p; do echo "$p yes;"; done < "$TMPD/new_prefixes" > "$TMPD/new_lines"
        head -n $((BING_LINE - 1)) "$WHITELIST_CONF" > "$TMPD/head"
        tail -n +$BING_LINE "$WHITELIST_CONF" > "$TMPD/rest"
        { cat "$TMPD/head"; cat "$TMPD/new_lines"; echo ""; cat "$TMPD/rest"; } > "$WHITELIST_CONF"
        echo "[OK] Added $NEW_COUNT new Google Bot IP prefixes"
    fi
else
    echo "[INFO] Google Bot IP whitelist is up to date (no new prefixes)"
fi

wget https://dl.miyuru.lk/geoip/maxmind/country/maxmind.dat.gz && gunzip maxmind.dat.gz && rm -f /vddos/GeoIP/GeoIP.dat && mv maxmind.dat /vddos/GeoIP/GeoIP.dat
wget https://dl.miyuru.lk/geoip/maxmind/asn/maxmind.dat.gz && gunzip maxmind.dat.gz && rm -f /vddos/GeoIP/GeoIPASN.dat && mv maxmind.dat /vddos/GeoIP/GeoIPASN.dat

# Add localhost IP only if it's a valid IP (not HTML or error)
if echo "$ipd" | grep -qE '^([0-9]{1,3}\.){3}[0-9]{1,3}$|^([0-9a-fA-F]{1,4}:){7}[0-9a-fA-F]{1,4}$'; then
    echo "# Localhost
$ipd yes;" >> /vddos/conf.d/whitelist-botsearch.conf
fi

# Xoa cac file JSON sau khi hoan tat (dung de tai lap khi chay lai)
for f in common-crawlers.json special-crawlers.json user-triggered-fetchers.json user-triggered-fetchers-google.json user-triggered-agents.json; do
    rm -f "$SCRIPT_DIR/$f"
done

vddos restart
