# -*- coding: utf-8 -*-
"""Gop IP tu 4 file JSON vao phan Google Bot cua whitelist-botsearch.conf (chi them IP chua co).
   Mac dinh tu dong tai 4 file JSON tu Google truoc khi xu ly. Dung --no-download neu da co file local."""
import argparse
import json
import os
import re
import ssl
import sys
import urllib.request

DIR = os.path.dirname(os.path.abspath(__file__))
CONF = os.path.join(DIR, "whitelist-botsearch.conf")

# URL chinh thuc tu Google (https://developers.google.com/crawling/ipranges)
JSON_URLS = {
    "common-crawlers.json": "https://developers.google.com/crawling/ipranges/common-crawlers.json",
    "special-crawlers.json": "https://developers.google.com/crawling/ipranges/special-crawlers.json",
    "user-triggered-fetchers.json": "https://developers.google.com/crawling/ipranges/user-triggered-fetchers.json",
    "user-triggered-fetchers-google.json": "https://developers.google.com/crawling/ipranges/user-triggered-fetchers-google.json",
}
JSON_FILES = list(JSON_URLS.keys())


def download_json_files(json_dir, skip_ssl_verify=False):
    """Tai 4 file JSON tu Google ve thu muc json_dir."""
    ctx = ssl.create_default_context()
    if skip_ssl_verify:
        ctx.check_hostname = False
        ctx.verify_mode = ssl.CERT_NONE
    for name, url in JSON_URLS.items():
        path = os.path.join(json_dir, name)
        try:
            req = urllib.request.Request(url, headers={"User-Agent": "WhitelistBotSearch/1.0"})
            with urllib.request.urlopen(req, timeout=30, context=ctx) as resp:
                data = resp.read().decode("utf-8")
            with open(path, "w", encoding="utf-8") as f:
                f.write(data)
            print("Da tai: %s" % name)
        except Exception as e:
            print("Loi khi tai %s: %s" % (url, e), file=sys.stderr)
            if not os.path.isfile(path):
                raise SystemExit(1)


def collect_from_json(json_dir):
    prefixes = set()
    for name in JSON_FILES:
        path = os.path.join(json_dir, name)
        if not os.path.isfile(path):
            continue
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        for p in data.get("prefixes", []):
            if "ipv4Prefix" in p:
                prefixes.add(("v4", p["ipv4Prefix"]))
            if "ipv6Prefix" in p:
                prefixes.add(("v6", p["ipv6Prefix"]))
    return prefixes

def get_existing_google_prefixes(conf_path):
    """Lay tat ca dong IP trong phan # Google Bot IP Addresses (tu dong 11 den truoc # Bing)."""
    with open(conf_path, "r", encoding="utf-8") as f:
        lines = f.readlines()
    existing = set()
    in_google = False
    for line in lines:
        s = line.strip()
        if s == "# Google Bot IP Addresses":
            in_google = True
            continue
        if in_google:
            if s.startswith("# ") and "Bot" in s:
                break
            # Dang "prefix yes;" hoac "prefix yes;"
            m = re.match(r"^([^\s#]+)\s+yes\s*;", s)
            if m:
                prefix = m.group(1).strip()
                existing.add(prefix)
    return existing

def main():
    parser = argparse.ArgumentParser(description="Cap nhat Google Bot IP vao whitelist-botsearch.conf")
    parser.add_argument("--no-download", action="store_true", help="Khong tai JSON tu Google, dung file local")
    parser.add_argument("--conf", default=CONF, help="Duong dan file whitelist-botsearch.conf (mac dinh: trong cung thu muc script)")
    parser.add_argument("--dir", default=DIR, help="Thu muc chua 4 file JSON / luu file tai ve (mac dinh: cung thu muc script)")
    parser.add_argument("--no-ssl-verify", action="store_true", help="Tat xac thuc SSL khi tai file (chi dung neu can)")
    args = parser.parse_args()
    conf_path = os.path.abspath(args.conf)
    json_dir = os.path.abspath(args.dir)

    if not args.no_download:
        print("Dang tai 4 file JSON tu Google...")
        download_json_files(json_dir, skip_ssl_verify=args.no_ssl_verify)
        print("")

    from_json = collect_from_json(json_dir)
    existing = get_existing_google_prefixes(conf_path)

    # Chi them prefix chua co trong whitelist; da co thi bo qua (tranh trung)
    new_v4 = []
    new_v6 = []
    for ver, prefix in from_json:
        if prefix not in existing:
            if ver == "v4":
                new_v4.append(prefix)
            else:
                new_v6.append(prefix)

    # Sort: IPv4 sort theo so, IPv6 sort theo string
    def ipv4_key(s):
        parts = s.replace("/", " ").split()
        addr = parts[0]
        cidr = int(parts[1]) if len(parts) > 1 else 0
        return tuple(int(x) for x in addr.split(".")) + (cidr,)

    new_v4.sort(key=ipv4_key)
    new_v6.sort()

    new_lines = [p + " yes;\n" for p in new_v4 + new_v6]
    if not new_lines:
        print("Khong co IP nao moi de them. Tat ca da co trong whitelist.")
        return

    # Doc file conf, chen new_lines truoc dong "# Bing Bot IP Addresses"
    with open(conf_path, "r", encoding="utf-8") as f:
        content = f.read()

    marker = "\n# Bing Bot IP Addresses"
    if marker not in content:
        print("Khong tim thay marker '# Bing Bot IP Addresses'")
        return

    insert_block = "".join(new_lines)
    # Chen truoc marker (tuc la sau newline cuoi cua phan Google)
    new_content = content.replace(
        marker,
        "\n" + insert_block.rstrip() + "\n" + marker,
        1,
    )
    with open(conf_path, "w", encoding="utf-8") as f:
        f.write(new_content)

    print("Da them %d IPv4 va %d IPv6 vao # Google Bot IP Addresses." % (len(new_v4), len(new_v6)))
    if new_v4:
        print("IPv4 moi (so luong):", len(new_v4))
        for p in new_v4[:10]:
            print("  ", p)
        if len(new_v4) > 10:
            print("  ... va %d dong nua" % (len(new_v4) - 10))
    if new_v6:
        print("IPv6 moi (so luong):", len(new_v6))
        for p in new_v6[:5]:
            print("  ", p)
        if len(new_v6) > 5:
            print("  ... va %d dong nua" % (len(new_v6) - 5))

if __name__ == "__main__":
    main()
